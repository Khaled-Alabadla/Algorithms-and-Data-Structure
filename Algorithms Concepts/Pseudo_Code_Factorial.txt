START

  INPUT n

  IF n < 0 THEN
    OUTPUT "Please enter a non-negative integer"
    EXIT
  END IF

  IF n equals 0 OR n equals 1 THEN
    Set result to 1
  ELSE
    Set result to 1
    FOR each number from 2 to n
      Set result to result * number
    END FOR
  END IF

  OUTPUT "The factorial of " + n + " is: " + result
  
END
